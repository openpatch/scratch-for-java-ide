import java.awt.image.BufferedImage;
import java.io.ByteArrayOutputStream;
import java.io.PrintStream;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.TimeUnit;
import java.util.regex.Matcher;
import java.util.regex.Pattern;
import javax.imageio.ImageIO;
import org.openpatch.scratch.Shader;
import org.openpatch.scratch.Sprite;
import org.openpatch.scratch.Stage;
import org.openpatch.scratch.Window;
import org.openpatch.scratch.extension.Extension;
import org.openpatch.scratch.extension.ExtensionContext;

/**
 * Renders an image through a Scratch for Java 7 shader (the body format: {@code costume},
 * {@code uv}, {@code color}, {@code time}, {@code resolution}) the way the library draws it - by the
 * library itself, in a hidden window: a sprite with the image as costume and the shader on it.
 * Written by Scratch for Java Studio and run in source-file mode with the library's all-jar.
 *
 * <p>Arguments: {@code frag vert|- image|- out width height frames seconds [name=v1,v2,...]...}.
 * Prints {@code ERROR frag <line> <message>} for each compile error, {@code FRAME <file>} per frame
 * and {@code OK} at the end; {@code UNIFORMS name:type:size ...} lists the shader's uniforms (as
 * declared in its source). With {@code --serve} it keeps running and reads one request per line
 * (the arguments, tab-separated, or {@code LIVE time file} followed by them), each answered and
 * followed by {@code END}. One helper draws one frame size: the window cannot change its size.
 *
 * <p>{@code time} is the game's time. A helper that keeps running cannot set it back, so a
 * request starts at the next whole second of game time: {@code fract(time)} is exactly what was
 * asked for, a longer period may be shifted by whole seconds. Each frame is taken at its exact
 * tick (the game runs fast, then pauses and steps up to it). 7.0 shaders are fragment shaders; a
 * vertex shader is reported as an error. A shader that does not compile gives its errors and no
 * {@code OK}.
 */
public class ShaderPreview {

  static final Pattern UNIFORM = Pattern.compile(
      "\\buniform\\s+(\\w+)\\s+(\\w+)\\s*(?:\\[\\s*(\\d+)\\s*\\])?\\s*;");
  // Mesa "0:12(3): error", NVIDIA "0(12) : error", Apple "ERROR: 0:12:"
  static final Pattern LINE = Pattern.compile("\\b\\d+[:(](\\d+)");

  static Window window;
  static Stage stage;
  static Sprite sprite;
  static String loadedKey;
  static int width;
  static int height;
  static final Frames FRAMES = new Frames();
  static final ByteArrayOutputStream ERR = new ByteArrayOutputStream();

  public static void main(String[] args) throws Exception {
    PrintStream console = System.err;
    // the library reports shader compile errors on stderr: collect them for the answer
    System.setErr(new PrintStream(ERR, true, StandardCharsets.UTF_8) {
      @Override
      public void write(byte[] b, int off, int len) {
        super.write(b, off, len);
        console.write(b, off, len);
      }
    });
    if (args.length > 0 && args[0].equals("--serve")) {
      var in = new java.io.BufferedReader(new java.io.InputStreamReader(System.in, StandardCharsets.UTF_8));
      String line;
      while ((line = in.readLine()) != null) {
        try {
          String[] request = line.split("\t");
          if (request[0].equals("LIVE")) {
            live(Double.parseDouble(request[1]), Path.of(request[2]),
                java.util.Arrays.copyOfRange(request, 3, request.length));
          } else {
            render(request);
          }
        } catch (Throwable t) {
          System.out.println("FAILED " + t);
        }
        System.out.println("END");
        System.out.flush();
      }
      System.exit(0);
    }
    try {
      render(args);
    } finally {
      System.exit(0);
    }
  }

  /** Frames at 0, seconds/frames, ... into out/frame-N.png. */
  static void render(String[] args) throws Exception {
    Request r = Request.of(args);
    if (!setUp(r)) {
      return;
    }
    long start = wholeSecond(FRAMES.tick());
    for (int i = 0; i < r.frames; i++) {
      Path file = r.out.resolve("frame-" + i + ".png");
      long target = start + Math.round(r.seconds * i / Math.max(1, r.frames) * 60);
      if (!FRAMES.save(target, file)) {
        System.out.println("FAILED the frame did not come");
        return;
      }
      System.out.println("FRAME " + file);
    }
    if (!errors(r)) {
      System.out.println("OK");
    }
  }

  /** The first tick at or after {@code tick} that is a whole second of game time. */
  static long wholeSecond(long tick) {
    return (tick + 59) / 60 * 60;
  }

  static long liveStart = -1;
  static String liveKey;

  /** One frame at time seconds into the preview (time runs on, it does not loop). */
  static void live(double time, Path file, String[] args) throws Exception {
    Request r = Request.of(args);
    if (!setUp(r)) {
      return;
    }
    String key = String.join("\t", args);
    long asked = Math.round(time * 60);
    if (!key.equals(liveKey) || liveStart + asked < FRAMES.tick()) {
      liveKey = key;
      // a whole second, so that time runs on from where it was asked for
      liveStart = wholeSecond(FRAMES.tick() - asked);
    }
    if (!FRAMES.save(liveStart + asked, file)) {
      System.out.println("FAILED the frame did not come");
      return;
    }
    System.out.println("FRAME " + file);
    if (!errors(r)) {
      System.out.println("OK");
    }
  }

  /** The window (first request), the sprite with the image and the shader with its uniforms. */
  static boolean setUp(Request r) throws Exception {
    if (r.vert != null) {
      System.out.println("ERROR vert 0 Scratch for Java 7 shaders are fragment shaders only: the vertex shader is not used.");
    }
    if (window == null) {
      width = r.width;
      height = r.height;
      System.setProperty("scratch4j.hidden", "true");
      Window.use(FRAMES);
      window = new Window(width, height);
      stage = new Stage();
      stage.setColor(new org.openpatch.scratch.Color(0, 0, 0, 0));
      window.setStage(stage);
      window.setGameSpeed(8); // the frames of a request come quickly
    } else if (r.width != width || r.height != height) {
      System.out.println("FAILED this helper draws " + width + "x" + height);
      return false;
    }
    Path image = r.image != null ? r.image : checkerboard(r.out);
    String source = Files.readString(r.frag);
    Map<String, String[]> types = uniforms(source);
    StringBuilder listed = new StringBuilder("UNIFORMS");
    types.forEach((name, t) -> listed.append(' ').append(name).append(':').append(t[0]).append(':').append(t[1]));
    System.out.println(listed);
    String key = r.frag + "|" + Files.getLastModifiedTime(r.frag) + "|" + image;
    ERR.reset();
    CompletableFuture<Void> done = new CompletableFuture<>();
    FRAMES.ctx.runOnGameThread(() -> {
      try {
        if (!key.equals(loadedKey)) {
          loadedKey = key;
          if (sprite != null) {
            stage.remove(sprite);
          }
          sprite = new Sprite("image", image.toString());
          // the image fills the frame, as the game would show it at that size
          sprite.setSize(100 * Math.min(width / sprite.getWidth(), height / sprite.getHeight()));
          sprite.getShaders().add("preview", r.frag.toString());
          sprite.getShaders().switchTo("preview");
          stage.add(sprite);
        }
        Shader shader = sprite.getShaders().get("preview");
        r.uniforms.forEach((name, values) -> set(shader, name, values, types.get(name)));
        done.complete(null);
      } catch (Throwable t) {
        done.completeExceptionally(t);
      }
    });
    done.get(30, TimeUnit.SECONDS);
    return true;
  }

  static void set(Shader shader, String name, double[] v, String[] type) {
    String t = type == null ? "float" : type[0];
    int size = type == null ? 1 : Integer.parseInt(type[1]);
    if (size > 1) {
      int per = switch (t) { case "vec2" -> 2; case "vec3" -> 3; case "vec4" -> 4; default -> 1; };
      shader.set(name, v, per);
    } else if (t.equals("int")) {
      shader.set(name, (int) Math.round(v[0]));
    } else if (t.equals("bool")) {
      shader.set(name, v[0] != 0);
    } else if (t.equals("vec2") && v.length >= 2) {
      shader.set(name, v[0], v[1]);
    } else if ((t.equals("vec3") || t.equals("vec4")) && v.length >= 3) {
      shader.set(name, v, t.equals("vec3") ? 3 : 4);
    } else {
      shader.set(name, v[0]);
    }
  }

  /** The compile errors the library printed for this shader, with the shader file's lines. */
  static boolean errors(Request r) {
    String err = ERR.toString(StandardCharsets.UTF_8);
    int at = err.indexOf("does not compile");
    if (at < 0) {
      return false;
    }
    boolean any = false;
    for (String line : err.substring(err.indexOf('\n', at) + 1).split("\n")) {
      if (line.isBlank() || line.startsWith("Scratch for Java")) {
        continue;
      }
      Matcher m = LINE.matcher(line);
      int number = m.find() ? Integer.parseInt(m.group(1)) : 0;
      boolean warning = line.toLowerCase(java.util.Locale.ROOT).contains("warning");
      String message = line.replaceFirst("^((ERROR|WARNING):\\s*)?\\d+[:(]\\d+\\)?(\\(\\d+\\))?\\s*:\\s*", "").strip();
      System.out.println((warning ? "WARNING" : "ERROR") + " frag " + number + " " + message);
      any |= !warning;
    }
    return any;
  }

  /** name -> {type, size}, as declared (the library's own uniforms are not declared by learners). */
  static Map<String, String[]> uniforms(String source) {
    Map<String, String[]> out = new LinkedHashMap<>();
    Matcher m = UNIFORM.matcher(source.replaceAll("//[^\\n]*|/\\*(?s:.*?)\\*/", ""));
    while (m.find()) {
      out.put(m.group(2), new String[] {m.group(1), m.group(3) == null ? "1" : m.group(3)});
    }
    return out;
  }

  /** A grey checkerboard when no image is given. */
  static Path checkerboard(Path out) throws Exception {
    Path file = out.resolve("checkerboard.png");
    if (!Files.isRegularFile(file)) {
      BufferedImage image = new BufferedImage(width, height, BufferedImage.TYPE_INT_ARGB);
      for (int y = 0; y < height; y++) {
        for (int x = 0; x < width; x++) {
          image.setRGB(x, y, ((x / 8 + y / 8) % 2 == 0) ? 0xffcccccc : 0xff999999);
        }
      }
      Files.createDirectories(out);
      ImageIO.write(image, "png", file.toFile());
    }
    return file;
  }

  /** Saves the frame of a given tick, through the library's extension API. */
  static final class Frames implements Extension {
    volatile ExtensionContext ctx;
    private volatile long target = -1;
    private volatile Path file;
    private volatile CompletableFuture<Boolean> saved;

    @Override
    public String id() {
      return "org.openpatch.scratch4j.shader-preview";
    }

    @Override
    public String name() {
      return "Shader preview";
    }

    @Override
    public String version() {
      return "1";
    }

    @Override
    public void install(ExtensionContext context) {
      this.ctx = context;
      context.afterFrame((s, frame) -> {
        CompletableFuture<Boolean> waiting = this.saved;
        if (waiting == null) {
          return;
        }
        long left = this.target - context.getTickCount();
        if (left <= 0) {
          this.saved = null;
          window.resume();
          try {
            Files.createDirectories(this.file.getParent());
            frame.savePng(this.file);
            waiting.complete(true);
          } catch (Exception e) {
            waiting.completeExceptionally(e);
          }
        } else if (left <= 24) {
          // a frame runs up to 8 ticks, and a pause takes effect with the next one: from here
          // on one tick a frame, so the picture is the asked-for tick exactly
          if (!window.isPaused()) {
            window.pause();
          } else {
            window.step();
          }
        }
      });
    }

    long tick() {
      while (this.ctx == null) {
        Thread.onSpinWait();
      }
      return this.ctx.getTickCount();
    }

    boolean save(long tick, Path into) throws Exception {
      this.file = into;
      this.target = tick;
      CompletableFuture<Boolean> waiting = new CompletableFuture<>();
      this.saved = waiting;
      return waiting.get(30, TimeUnit.SECONDS);
    }
  }

  record Request(Path frag, Path vert, Path image, Path out, int width, int height, int frames,
      double seconds, Map<String, double[]> uniforms) {
    static Request of(String[] a) {
      Map<String, double[]> uniforms = new LinkedHashMap<>();
      for (int i = 8; i < a.length; i++) {
        int eq = a[i].indexOf('=');
        if (eq > 0) {
          String[] parts = a[i].substring(eq + 1).split(",");
          double[] values = new double[parts.length];
          for (int k = 0; k < parts.length; k++) {
            values[k] = Double.parseDouble(parts[k]);
          }
          uniforms.put(a[i].substring(0, eq), values);
        }
      }
      return new Request(Path.of(a[0]), a[1].equals("-") ? null : Path.of(a[1]),
          a[2].equals("-") ? null : Path.of(a[2]), Path.of(a[3]), Integer.parseInt(a[4]),
          Integer.parseInt(a[5]), Integer.parseInt(a[6]), Double.parseDouble(a[7]), uniforms);
    }
  }
}
